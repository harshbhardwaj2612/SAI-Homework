Solution: 
 
echo "mkdir service" > create-service
echo "cd service" >> create-service
echo "mkdir controllers" >> create-service
echo "mkdir models" >> create-service
echo "mkdir routes" >> create-service
echo "mkdir middleware" >> create-service
printf "echo 'console.log(" >> create-service
printf '"Server started");' >> create-service
echo "' > server.js" >> create-service
echo "echo 'API Service' > README.md" >> create-service
chmod +x create-service
create-service
